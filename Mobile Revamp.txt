Step 3 — mobile/src/components/GroupCard.tsx (full rewrite)
tsximport React from 'react';
import { View, Text, TouchableOpacity, StyleSheet } from 'react-native';
import { LinearGradient } from 'expo-linear-gradient';
import { Crown, ArrowRight } from 'lucide-react-native';
import { GroupListItem, UserBalance } from '../services/api';
import { useTheme } from '../context/ThemeContext';
import { formatCurrency } from '../utils/formatCurrency';
import CharacterShape from './CharacterShape';

interface GroupCardProps {
  group: GroupListItem;
  members?: UserBalance[];
  myNetBalance?: number;
  onPress: () => void;
  compact?: boolean;
}

const ROTATIONS = [-8, 4, 6, -10];
const SETTLED_THRESHOLD = 0.01;

export default function GroupCard({
  group, members = [], myNetBalance = 0, onPress, compact = false,
}: GroupCardProps) {
  const { colors } = useTheme();
  const visibleMembers = members.slice(0, 4);
  const balance = myNetBalance ?? 0;
  const isOwe  = balance < -SETTLED_THRESHOLD;
  const isOwed = balance >  SETTLED_THRESHOLD;
  const balanceColor = isOwe ? colors.warningBright : colors.accent;
  const cardW = compact ? 222 : ('100%' as any);

  return (
    <TouchableOpacity onPress={onPress} activeOpacity={0.88}
      style={[styles.card, { width: cardW }]}>
      <LinearGradient
        colors={colors.groupCardBg as [string, string, string, string]}
        locations={[0, 0.30, 0.56, 1]}
        start={{ x: 0.5, y: 0 }}
        end={{ x: 0.5, y: 1 }}
        style={StyleSheet.absoluteFillObject}
      />

      {/* Characters */}
      <View style={[styles.facesRow, compact && styles.facesRowCompact]}>
        {visibleMembers.map((m, i) => (
          <View key={m.user_id} style={{ alignItems: 'center' }}>
            {m.user_id === group.created_by
              ? <Crown size={11} color="#F4B400" style={{ marginBottom: 2 }} />
              : <View style={{ height: 12 }} />}
            <Text style={[styles.nick, { transform: [{ rotate: `${ROTATIONS[i] ?? 0}deg` }] }]}>
              {(m.character_nickname ?? m.name.split(' ')[0]).slice(0, 7)}
            </Text>
            <View style={{ transform: [{ rotate: `${(ROTATIONS[i] ?? 0) * 0.5}deg` }] }}>
              <CharacterShape
                shape={m.character_shape ?? 'rect'}
                color={m.character_color ?? '#6B7280'}
                variant={compact ? 'mini' : 'card'}
              />
            </View>
          </View>
        ))}
      </View>

      {/* Name pill */}
      <View style={styles.pillWrap}>
        <View style={[styles.pill, { backgroundColor: colors.groupPillBg }]}>
          <Text style={[styles.pillText, { color: colors.groupPillText }]} numberOfLines={1}>
            {group.name}
          </Text>
        </View>
      </View>

      {/* Stats */}
      <View style={[styles.stats, compact && styles.statsCompact]}>
        <View style={styles.statBlock}>
          <Text style={[styles.statLabel, { color: colors.groupLabelText }]}>TOTAL EXPENSES</Text>
          <View style={[styles.statBox, { backgroundColor: colors.groupStatBg }]}>
            <Text style={[styles.statVal, { color: colors.groupStatText, fontSize: compact ? 21 : 26 }]}>
              ${formatCurrency(group.total_expenses)}
            </Text>
          </View>
        </View>

        {members.length > 0 && (
          <View style={styles.statBlock}>
            <Text style={[styles.statLabel, {
              color: isOwe ? colors.warningBright : colors.groupLabelText
            }]}>
              {isOwed ? "YOU'RE OWED" : isOwe ? 'YOU OWE' : 'STATUS'}
            </Text>
            <View style={[styles.balanceBox, { backgroundColor: colors.groupStatBg }]}>
              <Text style={[styles.balanceVal, { color: balanceColor, fontSize: compact ? 21 : 26 }]}>
                {!isOwe && !isOwed ? '✓ Settled' : `$${formatCurrency(Math.abs(balance))}`}
              </Text>
              <View style={[styles.arrowCircle, { borderColor: balanceColor }]}>
                <ArrowRight size={compact ? 14 : 17} color={balanceColor} />
              </View>
            </View>
          </View>
        )}
      </View>
    </TouchableOpacity>
  );
}

const styles = StyleSheet.create({
  card: { borderRadius: 28, overflow: 'hidden' },
  facesRow: { flexDirection: 'row', justifyContent: 'center', alignItems: 'flex-end',
    gap: 2, paddingTop: 20, paddingHorizontal: 16, zIndex: 1 },
  facesRowCompact: { paddingTop: 16, paddingHorizontal: 12 },
  nick: { fontSize: 10, fontWeight: '700', color: '#0E140F', marginBottom: 3,
    textShadowColor: 'rgba(255,255,255,0.55)', textShadowOffset: { width: 0, height: 1 }, textShadowRadius: 2 },
  pillWrap: { alignItems: 'center', marginTop: -17, paddingHorizontal: 14, zIndex: 2 },
  pill: { borderRadius: 999, paddingHorizontal: 24, paddingVertical: 11, maxWidth: '92%',
    shadowColor: '#000', shadowOffset: { width: 0, height: 6 }, shadowOpacity: 0.25, shadowRadius: 8, elevation: 6 },
  pillText: { fontSize: 22, fontWeight: '800', letterSpacing: -0.4 },
  stats: { padding: 14, paddingHorizontal: 18, paddingBottom: 18, gap: 10 },
  statsCompact: { padding: 12, paddingHorizontal: 14, paddingBottom: 14 },
  statBlock: { gap: 6 },
  statLabel: { fontSize: 10, fontWeight: '800', letterSpacing: 1.3, textTransform: 'uppercase', textAlign: 'center' },
  statBox: { borderRadius: 20, paddingVertical: 12, paddingHorizontal: 20,
    alignItems: 'center', justifyContent: 'center',
    shadowColor: '#000', shadowOffset: { width: 0, height: 6 }, shadowOpacity: 0.12, shadowRadius: 10, elevation: 3 },
  statVal: { fontWeight: '800', letterSpacing: -0.4 },
  balanceBox: { borderRadius: 20, paddingLeft: 20, paddingRight: 10, paddingVertical: 10,

  balanceVal: { fontWeight: '800', letterSpacing: -0.4 },
